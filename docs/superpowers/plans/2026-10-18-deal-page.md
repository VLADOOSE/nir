# Единая страница сделки — план реализации

> **Исполнение:** самому, в основной сессии, задача за задачей (superpowers:executing-plans) — субагентов не запускать (CLAUDE.md §2–3, решение оператора 2026-10-18). Шаги — чекбоксы `- [ ]`.

**Цель:** на странице частной заявки и в карточке тендера — этапы всего цикла с подсказкой «Дальше» и колонка «Переписка и история» (лента событий сделки).

**Архитектура:** этапы и «Дальше» считает сервер (`PrivateRequestProgressBuilder` поверх существующего `PrivateRequestStageCalculator`, `TenderProgressBuilder` поверх данных тендера). Лента собирается одним сервисом `DealTimelineService` из существующих таблиц (переписка обращения, запросы цен, письма, файлы ответов, КП, заявки на участие) плюс новая таблица `deal_event` (V32) для заметок, звонков и смены этапа. Фронт — три общих компонента (`deal-layout`, `deal-progress`, `deal-timeline`), встраиваются в страницу заявки и карточку тендера.

**Стек:** Spring Boot 3.5.6 / Java 17 / Spring Data JPA / Flyway (V32) / PostgreSQL 17; Angular 21 standalone-компоненты с инлайн-шаблонами.

**Спека:** `docs/superpowers/specs/2026-10-18-deal-page-design.md` (читать вместе с планом).

package com.vladoose.nir.service.deal;

import com.vladoose.nir.entity.DealEvent;
import com.vladoose.nir.entity.DealEventType;
import com.vladoose.nir.exception.BadRequestException;
import com.vladoose.nir.repository.DealEventRepository;
import com.vladoose.nir.util.CurrentUser;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/** Запись событий сделки (спека 2026-10-18 §5). Гард рынка — у вызывающего (заявка / тендер уже проверены). */
@Service
public class DealEventService {

    public static final int MAX_TEXT = 2000;
    public static final String OFFER_STATUS = "OFFER_STATUS";
    public static final String CLOSED = "CLOSED";
    public static final String REOPENED = "REOPENED";
    public static final String APPLY_STATUS = "APPLY_STATUS";
    public static final String WINNER = "WINNER";

    private final DealEventRepository repository;

    public DealEventService(DealEventRepository repository) {
        this.repository = repository;
    }

    /** Заметка или звонок: текст обязателен, до 2000 символов, без пробелов по краям. */
    @Transactional
    public DealEvent note(Long tenderId, DealEventType type, String text) {
        if (type != DealEventType.NOTE && type != DealEventType.CALL) {
            throw new BadRequestException("Событие — заметка или звонок");
        }
        String t = text == null ? "" : text.strip();
        if (t.isEmpty()) throw new BadRequestException("Текст пустой");
        if (t.length() > MAX_TEXT) throw new BadRequestException("Текст длиннее " + MAX_TEXT + " символов");
        return repository.save(DealEvent.builder().tenderId(tenderId).type(type).text(t).author(CurrentUser.name()).build());
    }

    /** Смена этапа — в транзакции действия: откатилось действие — откатилось и событие. */
    @Transactional
    public DealEvent stage(Long tenderId, String code, String text, Long refId) {
        return repository.save(DealEvent.builder().tenderId(tenderId).type(DealEventType.STAGE).code(code).text(text)
                .refId(refId).author(CurrentUser.name()).build());
    }
}

public final class DealTexts {
    private static final DateTimeFormatter DD_MM = DateTimeFormatter.ofPattern("dd.MM");
    private static final DateTimeFormatter DD_MM_HH = DateTimeFormatter.ofPattern("dd.MM HH:mm");

    private DealTexts() {}

    public static String ddMM(OffsetDateTime at, ZoneId zone) { return at == null ? null : DD_MM.format(at.atZoneSameInstant(zone)); }
    public static String ddMM(LocalDate d) { return d == null ? null : DD_MM.format(d); }
    public static String ddMMhhmm(LocalDateTime at) { return at == null ? null : DD_MM_HH.format(at); }

    /** 1 позиция, 2 позиции, 5 позиций, 11 позиций, 21 позиция. */
    public static String plural(long n, String one, String few, String many) {
        long m10 = n % 10, m100 = n % 100;
        if (m10 == 1 && m100 != 11) return one;
        if (m10 >= 2 && m10 <= 4 && (m100 < 12 || m100 > 14)) return few;
        return many;
    }

    public static String positions(long n) { return n + " " + plural(n, "позиция", "позиции", "позиций"); }
    public static String lots(long n) { return n + " " + plural(n, "лот", "лота", "лотов"); }

    /** «ответил 1 из 3» / «ответили 2 из 5»; ноль — «ответов пока нет». */
    public static String answered(long answered, long of, boolean capital) {
        String s = answered == 0
                ? "ответов пока нет — запрошено у " + of + " " + plural(of, "поставщика", "поставщиков", "поставщиков")
                : plural(answered, "ответил", "ответили", "ответили") + " " + answered + " из " + of;
        return capital ? Character.toUpperCase(s.charAt(0)) + s.substring(1) : s;
    }

        return n + " " + plural(n, "письмо не ушло", "письма не ушли", "писем не ушли");

        long lines = stage.lineCount();

                        .map(e -> DealTexts.ddMM(e.getOccurredAt(), zone)).orElse(null);

            case "EMPTY" -> 1;
            case "NEW", "REQUESTED" -> 2;
            case "PRICED", "KP_DRAFT", "KP_SENT", "KP_REJECTED" -> 3;
            case "KP_ACCEPTED" -> 4;
            default -> 5;   // WON / LOST / CANCELLED — всё сделано

                : priced > 0 ? "цены по " + priced + " из " + lines

                : DealTexts.answered(answered, asked.size(), false);
        List<Step> steps = List.of(
                step("SOURCE", 0, phase, lead != null ? "Обращение" : "Создана",
                        lead != null ? DealTexts.channel(lead.getChannel()) + " " + DealTexts.ddMM(lead.getReceivedAt(), zone)
                                : DealTexts.ddMM(t.getCreatedAt(), zone)),
                step("LINES", 1, phase, "Позиции", lines == 0 ? "нет позиций" : DealTexts.positions(lines)),

                    + " — что ответил клиент?", "OFFER_DECISION", main.getId());

                    + " — новое КП или закройте сделку", "CREATE_OFFER", null);
            case "KP_ACCEPTED" -> new Next("Клиент согласился — закройте сделку", "CLOSE_WON", null);
            default -> new Next("Закрыта: " + outcome(status, null) + (closedAt == null ? "" : ", " + closedAt), null, null);
        };

    private static Step step(String key, int index, int phase, String label, String detail) {
        String state = index < phase ? "DONE" : index == phase ? "CURRENT" : "TODO";
        return new Step(key, state, label, detail);
    }

    /** Главное КП: принятое → последнее отправленное → последний черновик → последнее (отклонённое). */
    static ClientOffer mainOffer(List<ClientOffer> newestFirst) {
        for (ClientOfferStatus s : List.of(ClientOfferStatus.ACCEPTED, ClientOfferStatus.SENT, ClientOfferStatus.DRAFT)) {
            for (ClientOffer o : newestFirst) if (o.getStatus() == s) return o;
        }
        return newestFirst.isEmpty() ? null : newestFirst.get(0);
    }

    private static String offerState(ClientOffer o, ZoneId zone) {
        return switch (o.getStatus()) {
            case DRAFT -> "черновик";
            case SENT -> "отправлено" + (o.getSentAt() == null ? "" : " " + DealTexts.ddMM(o.getSentAt(), zone));
            case ACCEPTED -> "принято";
            case REJECTED -> "отклонено";
        };
    }

    private static String outcome(PrivateRequestStatus status, String closedAt) {
        String text = switch (status) {
            case WON -> "купили у нас";
            case LOST -> "не купили";
            case CANCELLED -> "отменена";
            case NEW -> null;
        };
        return text == null || closedAt == null ? text : text + " " + closedAt;

package com.vladoose.nir.service.deal;

import com.vladoose.nir.context.MarketContext;
import com.vladoose.nir.dto.response.ChatMessageResponse;

import com.vladoose.nir.dto.response.DealTimelineResponse;
import com.vladoose.nir.entity.*;

import com.vladoose.nir.repository.*;

import com.vladoose.nir.service.PrivateRequestService;
import org.springframework.data.domain.PageRequest;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.sql.Timestamp;
import java.time.*;
import java.time.temporal.ChronoUnit;
import java.util.*;

import java.util.stream.Stream;

/**
 * Лента «Переписка и история» (спека 2026-10-18 §3.2, §4.2): всё, у чего уже есть дата, — из существующих таблиц,
 * заметки / звонки / смена этапа — из deal_event. Страница — limit самых поздних раньше курсора, по возрастанию.

@Service
public class DealTimelineService {

    public static final int DEFAULT_LIMIT = 50;
    static final int MAX_LIMIT = 100;
    /** «Без границы» — дальше любой настоящей даты, в пределах timestamptz. */
    static final OffsetDateTime FAR = OffsetDateTime.of(9999, 1, 1, 0, 0, 0, 0, ZoneOffset.UTC);
    /** При равном времени событие раньше сообщения (как лента обращения). Сортировка списка устойчивая. */
    private static final Comparator<DealTimelineItem> ORDER = Comparator.comparing(DealTimelineItem::at)
            .thenComparing(i -> "MESSAGE".equals(i.kind()) ? 1 : 0);
    private static final Set<String> ANSWERED = Set.of("RESPONDED", "ACCEPTED", "DECLINED");

    private final PrivateRequestService privateRequests;
    private final TenderRepository tenders;
    private final LeadRepository leads;
    private final ChatMessageRepository messages;
    private final ChatService chats;
    private final PriceRequestRepository priceRequests;
    private final InboundEmailRepository inbound;
    private final SupplierReplyFileRepository replyFiles;
    private final ClientOfferRepository offers;
    private final ActivityApplyRepository applies;
    private final DealEventRepository events;

    public DealTimelineService(PrivateRequestService privateRequests, TenderRepository tenders, LeadRepository leads,
                               ChatMessageRepository messages, ChatService chats, PriceRequestRepository priceRequests,
                               InboundEmailRepository inbound, SupplierReplyFileRepository replyFiles,
                               ClientOfferRepository offers, ActivityApplyRepository applies, DealEventRepository events) {
        this.privateRequests = privateRequests;
        this.tenders = tenders;
        this.leads = leads;
        this.messages = messages;
        this.chats = chats;
        this.priceRequests = priceRequests;
        this.inbound = inbound;
        this.replyFiles = replyFiles;
        this.offers = offers;
        this.applies = applies;
        this.events = events;
    }

    /** Тендер (не частная заявка) текущего рынка, иначе 404: findById фильтром рынка не режется (CLAUDE.md §14). */
    public Tender publicTender(Long id) {
        Tender t = tenders.findById(id).orElseThrow(() -> new NotFoundException("Тендер не найден: id=" + id));
        if (t.getSource() == Source.PRIVATE_REQUEST || (t.getMarket() != null && t.getMarket() != MarketContext.get())) {
            throw new NotFoundException("Тендер не найден: id=" + id);
        }
        return t;
    }

    @Transactional(readOnly = true)
    public DealTimelineResponse forPrivateRequest(Long id, OffsetDateTime before, int limit) {
        Tender t = privateRequests.findById(id);
        OffsetDateTime cursor = before == null ? FAR : before;
        int n = clamp(limit);
        List<DealTimelineItem> all = new ArrayList<>();
        leads.findFirstByPrivateRequestId(t.getId()).ifPresent(lead -> addLead(all, lead, cursor, n));
        all.add(item(t.getCreatedAt(), "CREATED", "Создана заявка " + t.getTenderNumber(), null, null, null));
        addSupply(all, t, zone(t));
        addOffers(all, t.getId());
        addEvents(all, t.getId());
        return page(all, cursor, n);
    }

    public DealTimelineItem toItem(DealEvent e) {
        String link = e.getCode() == null ? null : switch (e.getCode()) {
            case DealEventService.OFFER_STATUS -> "OFFER";
            case DealEventService.APPLY_STATUS -> "APPLY";
            default -> null;
        };
        return new DealTimelineItem(e.getOccurredAt(), false, e.getType().name(), e.getText(), e.getAuthor(), link,
                link == null ? null : e.getRefId(), null);
    }

    private void addLead(List<DealTimelineItem> all, Lead lead, OffsetDateTime cursor, int n) {
        OffsetDateTime start = lead.getReceivedAt() != null ? lead.getReceivedAt() : lead.getCreatedAt();
        all.add(item(start, "LEAD", "Обращение (" + DealTexts.channel(lead.getChannel()) + ")"
                + (blank(lead.getSubject()) ? "" : ": «" + lead.getSubject().strip() + "»"), null, "LEAD", lead.getId()));
        for (LeadEvent e : lead.getEvents()) {
            String kind = switch (e.getType()) {
                case NOTE -> "LEAD_NOTE";
                case CALL -> "LEAD_CALL";
                case STATUS -> "LEAD_STATUS";
                default -> null;   // RECEIVED — строка «Обращение» выше; SYNC — служебное
            };
            if (kind != null) all.add(item(e.getOccurredAt(), kind, e.getBody(), e.getAuthor(), "LEAD", lead.getId()));
        }
        if (lead.getChat() == null || start == null) return;
        Long chatId = lead.getChat().getId();
        OffsetDateTime to = leads.findFirstByChatIdAndReceivedAtAfterOrderByReceivedAtAsc(chatId, start)
                .map(Lead::getReceivedAt).orElse(FAR);
        List<ChatMessage> newestFirst = messages.findForDeal(chatId, start, to, cursor, PageRequest.of(0, n + 1));
        for (ChatMessageResponse m : chats.toResponses(newestFirst)) {
            all.add(new DealTimelineItem(m.getSentAt(), false, "MESSAGE", null, null, "CHAT", chatId, m));
        }
    }

    private void addSupply(List<DealTimelineItem> all, Tender t, ZoneId zone) {
        List<PriceRequest> prs = priceRequests.findByTenderId(t.getId());
        if (prs.isEmpty()) return;
        Map<Long, PriceRequest> byId = new HashMap<>();
        prs.forEach(pr -> byId.put(pr.getId(), pr));

        Map<OffsetDateTime, List<PriceRequest>> sent = new TreeMap<>();   // одно окно «Запросить цены» — одна строка
        for (PriceRequest pr : prs) {
            LetterStatus ls = pr.getLetterStatus();
            if (pr.getSentAt() != null && (ls == null || ls == LetterStatus.SENT || ls == LetterStatus.MANUAL)) {
                sent.computeIfAbsent(pr.getSentAt().truncatedTo(ChronoUnit.MINUTES), k -> new ArrayList<>()).add(pr);
            }
            if (ls == LetterStatus.NO_EMAIL || ls == LetterStatus.FAILED) {
                all.add(item(pr.getSentAt() != null ? pr.getSentAt() : pr.getCreatedAt(), "LETTER_FAILED",
                        name(pr) + ": письмо не ушло — " + (ls == LetterStatus.NO_EMAIL ? "нет почты" : "ошибка отправки"),
                        null, "PRICE_REQUEST", pr.getId()));
            }
        }
        for (List<PriceRequest> group : sent.values()) {
            String mail = group.stream().filter(pr -> pr.getLetterStatus() != LetterStatus.MANUAL)
                    .map(DealTimelineService::name).collect(Collectors.joining(", "));
            String manual = group.stream().filter(pr -> pr.getLetterStatus() == LetterStatus.MANUAL)
                    .map(DealTimelineService::name).collect(Collectors.joining(", "));
            String text = "Запрос цен: " + Stream.of(mail.isEmpty() ? null : "письмом — " + mail,
                            manual.isEmpty() ? null : "передан вручную — " + manual)
                    .filter(Objects::nonNull).collect(Collectors.joining("; "));
            OffsetDateTime at = group.stream().map(PriceRequest::getSentAt).min(Comparator.naturalOrder()).orElseThrow();
            boolean one = group.size() == 1;
            all.add(item(at, "RFQ_SENT", text, null, one ? "PRICE_REQUEST" : null, one ? group.get(0).getId() : null));
        }

        Map<Long, OffsetDateTime> lastReply = new HashMap<>();
        for (Object[] row : inbound.findDealRows(byId.keySet())) {
            PriceRequest pr = byId.get(((Number) row[0]).longValue());
            OffsetDateTime at = offset(row[2]);
            if (pr == null || at == null) continue;
            switch ((String) row[1]) {
                case "SUPPLIER_RESPONSE" -> lastReply.merge(pr.getId(), at, (x, y) -> x.isAfter(y) ? x : y);
                case "BOUNCE" -> all.add(item(at, "BOUNCED", name(pr) + ": письмо вернулось", null, "PRICE_REQUEST", pr.getId()));
                case "DELAYED" -> all.add(item(at, "DELAYED", name(pr) + ": письмо задерживается", null, "PRICE_REQUEST", pr.getId()));
                default -> { }
            }
        }
        for (PriceRequest pr : prs) {
            if (!ANSWERED.contains(pr.getStatus())) continue;
            OffsetDateTime at = lastReply.get(pr.getId());
            boolean dateOnly = at == null;
            if (dateOnly) {
                if (pr.getResponseDate() == null) continue;
                at = pr.getResponseDate().atTime(LocalTime.NOON).atZone(zone).toOffsetDateTime();
            }
            all.add(new DealTimelineItem(at, dateOnly, "DECLINED".equals(pr.getStatus()) ? "DECLINED" : "REPLY",
                    replyText(pr), null, "PRICE_REQUEST", pr.getId(), null));
        }
        for (Object[] row : replyFiles.findDealRows(byId.keySet())) {
            PriceRequest pr = byId.get(((Number) row[0]).longValue());
            OffsetDateTime at = offset(row[2]);
            if (pr == null || at == null) continue;
            all.add(item(at, "REPLY_FILE", "Файл «" + row[1] + "» от " + name(pr), null, "PRICE_REQUEST", pr.getId()));
        }
    }

    private void addOffers(List<DealTimelineItem> all, Long tenderId) {
        for (ClientOffer o : offers.findByTenderIdOrderByIdDesc(tenderId)) {
            all.add(item(o.getCreatedAt(), "OFFER_CREATED", "КП № " + o.getNumber() + " создано", o.getCreatedBy(),
                    "OFFER", o.getId()));
            if (o.getSentAt() != null) {
                all.add(item(o.getSentAt(), "OFFER_SENT", "КП № " + o.getNumber() + " отправлено", null, "OFFER", o.getId()));
            }
        }
    }

    private void addEvents(List<DealTimelineItem> all, Long tenderId) {
        for (DealEvent e : events.findByTenderIdOrderByOccurredAtAscIdAsc(tenderId)) all.add(toItem(e));
    }

    static DealTimelineResponse page(List<DealTimelineItem> all, OffsetDateTime cursor, int n) {
        List<DealTimelineItem> eligible = all.stream().filter(i -> i.at() != null && i.at().isBefore(cursor))
                .sorted(ORDER).toList();
        int from = Math.max(0, eligible.size() - n);
        return new DealTimelineResponse(new ArrayList<>(eligible.subList(from, eligible.size())), from > 0);
    }

    static int clamp(int limit) {
        return limit <= 0 ? DEFAULT_LIMIT : Math.min(limit, MAX_LIMIT);
    }

    static ZoneId zone(Tender t) {
        return (t.getMarket() == null ? MarketContext.get() : t.getMarket()).zone();
    }

    static DealTimelineItem item(OffsetDateTime at, String kind, String text, String author, String linkType, Long linkId) {
        return new DealTimelineItem(at, false, kind, text, author, linkType, linkId, null);
    }

    private static String replyText(PriceRequest pr) {
        if ("DECLINED".equals(pr.getStatus())) return name(pr) + " отказался";
        long asked = pr.getItems().size();
        long priced = pr.getItems().stream().filter(i -> i.getResponsePrice() != null).count();
        return priced > 0 ? name(pr) + " ответил: цены по " + priced + " из " + asked : name(pr) + " ответил — цен нет";
    }

    static String name(PriceRequest pr) {
        return pr.getDistributor() == null ? "Поставщик" : pr.getDistributor().getName();
    }

    /** Время из нативного запроса: Hibernate 6 отдаёт timestamptz как OffsetDateTime или Instant, драйвер — Timestamp. */
    static OffsetDateTime offset(Object v) {
        if (v instanceof OffsetDateTime o) return o;
        if (v instanceof Instant i) return i.atOffset(ZoneOffset.UTC);
        if (v instanceof Timestamp ts) return ts.toInstant().atOffset(ZoneOffset.UTC);
        if (v instanceof LocalDateTime l) return l.atOffset(ZoneOffset.UTC);
        return null;
    }

    private static boolean blank(String s) {
        return s == null || s.isBlank();
    }
}

package com.vladoose.nir.service.deal;

import com.vladoose.nir.dto.response.DealProgress;
import com.vladoose.nir.dto.response.DealProgress.Next;
import com.vladoose.nir.dto.response.DealProgress.Step;
import com.vladoose.nir.entity.*;
import com.vladoose.nir.integration.ImportDeadline;
import com.vladoose.nir.repository.ActivityApplyRepository;
import com.vladoose.nir.repository.PriceRequestItemRepository;
import com.vladoose.nir.repository.PriceRequestRepository;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.*;

/** Этапы и «Дальше» карточки тендера (спека 2026-10-18 §4.1). now — местное время рынка (срок площадки — без пояса). */
@Component
public class TenderProgressBuilder {

    private static final Set<String> ANSWERED = Set.of("RESPONDED", "ACCEPTED", "DECLINED");
    private static final String[] KEYS = {"LOTS", "PRICES", "APPLY", "RESULT", "CONTRACT", "DELIVERY", "PAYMENT"};
    private static final String[] LABELS = {"Лоты и ТЗ", "Цены", "Заявка", "Итог", "Договор", "Поставка", "Оплата"};

    private final PriceRequestRepository priceRequests;
    private final PriceRequestItemRepository items;
    private final ActivityApplyRepository applies;

    public TenderProgressBuilder(PriceRequestRepository priceRequests, PriceRequestItemRepository items,
                                 ActivityApplyRepository applies) {
        this.priceRequests = priceRequests;
        this.items = items;
        this.applies = applies;
    }

    @Transactional(readOnly = true)
    public DealProgress build(Tender t, LocalDateTime now) {
        List<TenderLot> lots = t.getLots();
        boolean imported = t.getPlatform() != null;
        long withTz = lots.stream().filter(l -> l.getTechSpecStatus() == TechSpecStatus.OK).count();
        List<PriceRequest> prs = priceRequests.findByTenderId(t.getId());
        Map<Long, Boolean> asked = new HashMap<>();
        for (PriceRequest pr : prs) asked.merge(pr.getDistributor().getId(), ANSWERED.contains(pr.getStatus()), Boolean::logicalOr);
        long answered = asked.values().stream().filter(Boolean::booleanValue).count();
        int priced = items.findPricedLots(List.of(t.getId())).size();
        ActivityApply apply = mainApply(applies.findByTenderId(t.getId()));
        String a = apply == null ? null : apply.getStatus();
        boolean cancelled = "CANCELLED".equals(t.getStatus());
        boolean passed = ImportDeadline.passed(t.getDeadline(), t.getDeadlineAt(), now);
        String deadline = t.getDeadlineAt() != null ? DealTexts.ddMMhhmm(t.getDeadlineAt()) : DealTexts.ddMM(t.getDeadline());

        int phase;
        boolean ended = false;   // проиграли / отменено / срок упущен — хвост пропускается
        Next next;
        if ("WON".equals(a)) {
            boolean contract = !blank(apply.getContractNumber()) || apply.getContractSignedAt() != null;
            boolean paid = apply.getPaidAt() != null || "PAID".equals(apply.getDeliveryStatus());
            boolean delivered = paid || apply.getDeliveredAt() != null || "DELIVERED".equals(apply.getDeliveryStatus());
            phase = !contract ? 4 : !delivered ? 5 : !paid ? 6 : 7;
            next = switch (phase) {
                case 4 -> new Next("Выиграли — заключите договор, затем поставка и оплата", "OPEN_APPLY", apply.getId());
                case 5 -> new Next("Договор есть — поставка" + ("ORDERED".equals(apply.getDeliveryStatus()) ? ": заказано" : ""),
                        "OPEN_APPLY", apply.getId());
                case 6 -> new Next("Поставлено — ждём оплату", "OPEN_APPLY", apply.getId());
                default -> new Next("Сделка закрыта: оплачено"
                        + (apply.getPaidAt() == null ? "" : " " + DealTexts.ddMM(apply.getPaidAt())), null, null);
            };
        } else if (cancelled || "LOST".equals(a) || "CANCELLED".equals(a) || "REJECTED".equals(a)) {
            phase = 3;
            ended = true;
            next = new Next(cancelled ? "Тендер отменён на площадке" : "LOST".equals(a) ? "Тендер проигран" : "Заявка отменена",
                    null, null);
        } else if ("SUBMITTED".equals(a) || "UNDER_REVIEW".equals(a)) {
            phase = 3;
            next = new Next("Ждём итога", null, null);
        } else if ("DRAFT".equals(a)) {
            phase = 2;
            ended = passed;
            next = passed ? closedNoApply(deadline)
                    : new Next("Подайте заявку" + (deadline == null ? "" : " — приём до " + deadline), "OPEN_APPLY", apply.getId());
        } else if (prs.isEmpty()) {
            boolean needTz = imported && withTz < lots.size();
            phase = needTz ? 0 : 1;
            ended = passed;
            next = passed ? closedNoApply(deadline)
                    : needTz ? new Next("Разберите ТЗ: " + DealTexts.lots(lots.size() - withTz) + " без ТЗ", "TO_LOTS", null)
                    : new Next("Запросите цены по лотам", "TO_LOTS", null);
        } else if (priced == 0) {
            phase = 1;
            ended = passed;
            next = passed ? closedNoApply(deadline) : new Next(DealTexts.answered(answered, asked.size(), true), "OPEN_REQUESTS", null);
        } else {
            phase = 2;
            ended = passed;
            next = passed ? closedNoApply(deadline)
                    : new Next("Цены есть — выберите победителей и соберите заявку", "OPEN_REQUESTS", null);
        }

        String[] details = {
                imported ? "ТЗ: " + withTz + " из " + lots.size() : DealTexts.lots(lots.size()),
                prs.isEmpty() ? "не запрашивали" : priced > 0 ? "цены по " + priced + " из " + lots.size()
                        : answered == 0 ? "ответов пока нет" : DealTexts.answered(answered, asked.size(), false),
                apply == null ? null : "№ " + apply.getId() + " · " + ApplyStatusLabels.label(a),
                result(a, cancelled),
                contract(apply),
                delivery(apply),
                payment(apply)};
        boolean lostRun = ended && phase == 3;
        List<Step> steps = new ArrayList<>();
        for (int i = 0; i < KEYS.length; i++) {
            String state = i < phase ? "DONE"
                    : i == phase ? (lostRun ? "DONE" : "CURRENT")
                    : ended ? "SKIPPED" : "TODO";
            steps.add(new Step(KEYS[i], state, LABELS[i], details[i]));
        }
        return new DealProgress(steps, next);
    }

    private static Next closedNoApply(String deadline) {
        return new Next("Приём закрыт" + (deadline == null ? "" : " " + deadline) + " — заявку не подали", null, null);
    }

    /** Последняя не отменённая, иначе последняя. */
    static ActivityApply mainApply(List<ActivityApply> list) {
        List<ActivityApply> sorted = new ArrayList<>(list);
        sorted.sort(Comparator.comparing(ActivityApply::getId).reversed());
        return sorted.stream().filter(x -> !"CANCELLED".equals(x.getStatus()) && !"REJECTED".equals(x.getStatus()))
                .findFirst().orElse(sorted.isEmpty() ? null : sorted.get(0));
    }

    private static String result(String a, boolean cancelled) {
        if ("WON".equals(a)) return "выиграли";
        if (cancelled) return "тендер отменён";
        if (a == null) return null;
        return switch (a) {
            case "SUBMITTED", "UNDER_REVIEW" -> "на рассмотрении";
            case "LOST" -> "проиграли";
            case "CANCELLED" -> "отменена";
            case "REJECTED" -> "отклонена";
            default -> null;
        };
    }

    private static String contract(ActivityApply a) {
        if (a == null || (blank(a.getContractNumber()) && a.getContractSignedAt() == null)) return null;
        String num = blank(a.getContractNumber()) ? "подписан" : "№ " + a.getContractNumber().strip();
        return num + (a.getContractSignedAt() == null ? "" : " от " + DealTexts.ddMM(a.getContractSignedAt()));
    }

    private static String delivery(ActivityApply a) {
        if (a == null) return null;
        if (a.getDeliveredAt() != null) return "доставлено " + DealTexts.ddMM(a.getDeliveredAt());
        if ("DELIVERED".equals(a.getDeliveryStatus()) || "PAID".equals(a.getDeliveryStatus())) return "доставлено";
        if ("ORDERED".equals(a.getDeliveryStatus())) return "заказано";
        return null;
    }

    private static String payment(ActivityApply a) {
        if (a == null) return null;
        if (a.getPaidAt() != null) return "оплачено " + DealTexts.ddMM(a.getPaidAt());
        return "PAID".equals(a.getDeliveryStatus()) ? "оплачено" : null;
    }

    private static boolean blank(String s) {
        return s == null || s.isBlank();
    }
}

    @Transactional(readOnly = true)
    public DealTimelineResponse forTender(Long id, OffsetDateTime before, int limit) {
        Tender t = publicTender(id);
        OffsetDateTime cursor = before == null ? FAR : before;
        int n = clamp(limit);
        ZoneId zone = zone(t);
        List<DealTimelineItem> all = new ArrayList<>();
        all.add(item(t.getCreatedAt(), "TENDER_FOUND", foundText(t), null, null, null));
        addSupply(all, t, zone);
        addApplies(all, t.getId(), zone);
        addEvents(all, t.getId());
        return page(all, cursor, n);
    }

    private static String foundText(Tender t) {
        if (t.getPlatform() == TenderPlatform.GOSZAKUP) return "Тендер найден на goszakup.gov.kz";
        if (t.getPlatform() == TenderPlatform.SK_PHARMACY) return "Тендер найден на СК-Фармации";
        return "Тендер создан вручную";
    }

    private void addApplies(List<DealTimelineItem> all, Long tenderId, ZoneId zone) {
        for (ActivityApply a : applies.findByTenderId(tenderId)) {
            all.add(item(a.getCreatedAt(), "APPLY_CREATED", "Заявка на участие № " + a.getId() + " создана", null,
                    "APPLY", a.getId()));
            if (a.getContractSignedAt() != null) {
                all.add(dateItem(a.getContractSignedAt(), zone, "CONTRACT", "Договор"
                        + (blank(a.getContractNumber()) ? "" : " № " + a.getContractNumber().strip()) + " подписан", a.getId()));
            }
            if (a.getDeliveredAt() != null) all.add(dateItem(a.getDeliveredAt(), zone, "DELIVERED", "Поставлено", a.getId()));
            if (a.getPaidAt() != null) all.add(dateItem(a.getPaidAt(), zone, "PAID", "Оплачено", a.getId()));
        }
    }

    private static DealTimelineItem dateItem(LocalDate d, ZoneId zone, String kind, String text, Long applyId) {
        return new DealTimelineItem(d.atTime(LocalTime.NOON).atZone(zone).toOffsetDateTime(), true, kind, text, null,
                "APPLY", applyId, null);
    }

  @Input() phone: string | null = null;
  @Input() email: string | null = null;
  @ViewChild('scroll') scrollRef?: ElementRef<HTMLElement>;
  @ViewChild('list') listRef?: ElementRef<HTMLElement>;

  items: DealTimelineItem[] = [];
  hasMore = false;
  loading = false;
  loadingOlder = false;
  error = '';
  panel: 'NOTE' | 'CALL' | null = null;
  draft = '';
  saving = false;
  private pinned = true;
  private sub?: Subscription;
  private timer?: ReturnType<typeof setInterval>;
  private ro?: ResizeObserver;
  private readonly onVisible = () => { if (document.visibilityState === 'visible') this.refresh(); };

  constructor(private api: ApiService, private notify: NotificationService, private cdr: ChangeDetectorRef) {}

  ngOnChanges(ch: SimpleChanges) {
    if (ch['dealId'] || ch['kind']) this.load();
  }

  ngAfterViewInit() {
    this.ro = new ResizeObserver(() => { if (this.pinned) this.stick(); });
    if (this.listRef) this.ro.observe(this.listRef.nativeElement);
    if (this.scrollRef) this.ro.observe(this.scrollRef.nativeElement);
    this.timer = setInterval(() => { if (document.visibilityState === 'visible') this.refresh(); }, 30_000);
    document.addEventListener('visibilitychange', this.onVisible);
  }

  ngOnDestroy() {
    this.sub?.unsubscribe();
    this.ro?.disconnect();
    if (this.timer) clearInterval(this.timer);
    document.removeEventListener('visibilitychange', this.onVisible);
  }

  /** Первая страница — самые новые события; лента прижата к низу. */
  load() {
    this.sub?.unsubscribe();
    this.loading = true;
    this.error = '';
    this.pinned = true;
    const id = this.dealId;
    this.sub = this.api.getDealTimeline(this.kind, id).subscribe({
      next: r => {
        if (id !== this.dealId) return;
        this.items = r.items;
        this.hasMore = r.hasMore;
        this.loading = false;
        this.cdr.markForCheck();
      },
      error: e => { this.loading = false; this.error = errorText(e); this.cdr.markForCheck(); },
    });
  }

  refresh() {
    if (this.loading) return;
    if (!this.items.length) { this.load(); return; }
    const id = this.dealId;
    this.api.getDealTimeline(this.kind, id).subscribe({
      next: r => {
        if (id !== this.dealId || !r.items.length) return;
        const from = new Date(r.items[0].at).getTime();
        const older = this.items.filter(i => new Date(i.at).getTime() < from);
        this.items = [...older, ...r.items];
        if (!older.length) this.hasMore = r.hasMore;
        this.cdr.markForCheck();
      },
      error: () => { /* тихо: следующий тик повторит */ },
    });
  }

  loadOlder() {
    if (!this.items.length || this.loadingOlder) return;
    const id = this.dealId;
    const el = this.scrollRef?.nativeElement;
    const fromBottom = el ? el.scrollHeight - el.scrollTop : 0;
    this.loadingOlder = true;
    this.pinned = false;
    this.api.getDealTimeline(this.kind, id, this.items[0].at).subscribe({
      next: r => {
        if (id !== this.dealId) return;
        this.items = [...r.items, ...this.items];
        this.hasMore = r.hasMore;
        this.loadingOlder = false;
        this.cdr.detectChanges();
        if (el) el.scrollTop = el.scrollHeight - fromBottom;   // держим место чтения
      },
      error: e => { this.loadingOlder = false; this.notify.error(errorText(e)); this.cdr.markForCheck(); },
    });
  }

  open(type: 'NOTE' | 'CALL') { this.panel = type; this.draft = ''; }

  save() {
    const text = this.draft.trim();
    const type = this.panel;
    if (!text || !type) return;
    this.saving = true;
    const id = this.dealId;
    this.api.addDealEvent(this.kind, id, type, text).subscribe({
      next: item => {
        this.saving = false;
        this.panel = null;
        this.draft = '';
        if (id === this.dealId) { this.items = [...this.items, item]; this.pinned = true; }
        this.cdr.markForCheck();
      },
      error: e => { this.saving = false; this.notify.error(errorText(e)); this.cdr.markForCheck(); },
    });
  }

  onScroll() {
    const el = this.scrollRef?.nativeElement;
    this.pinned = !el || el.scrollHeight - el.scrollTop - el.clientHeight < 80;
  }

  private stick() {
    const el = this.scrollRef?.nativeElement;
    if (el) el.scrollTop = el.scrollHeight;
  }

  writeLink(): { href: string; label: string } | null {
    let digits = (this.phone || '').replace(/\D/g, '');
    if (digits.length === 11 && digits.startsWith('8')) digits = '7' + digits.slice(1);
    if (digits.length >= 10) return { href: 'https://wa.me/' + digits, label: 'Написать в WhatsApp' };
    if (this.email) return { href: 'mailto:' + this.email, label: 'Написать письмо' };
    return null;
  }

  when(i: DealTimelineItem): string {
    const d = new Date(i.at);
    const p = (n: number) => String(n).padStart(2, '0');
    const day = p(d.getDate()) + '.' + p(d.getMonth() + 1);
    return i.dateOnly ? day : day + ' ' + p(d.getHours()) + ':' + p(d.getMinutes());
  }

  icon(kind: string) { return KIND_ICONS[kind] ?? '•'; }
  prefix(kind: string) { return kind === 'NOTE' || kind === 'LEAD_NOTE' ? 'Заметка:' : kind === 'CALL' || kind === 'LEAD_CALL' ? 'Звонок:' : null; }
  link(i: DealTimelineItem) { return timelineLink(i); }
  track(_: number, i: DealTimelineItem) { return i.kind + ':' + i.at + ':' + (i.message?.id ?? i.linkId ?? '') + ':' + (i.text ?? ''); }
}
